=================== ejecutar_recuperacion.sh
#!/usr/bin/env bash

set -Eeuo pipefail

ROOT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
cd "$ROOT_DIR"

PROJECT="dba-sgd-pitr-demo"
if [[ -f "$ROOT_DIR/.env" ]]; then
    PROJECT="$(sed -n 's/^COMPOSE_PROJECT_NAME=//p' "$ROOT_DIR/.env" | tail -n 1)"
    PROJECT="${PROJECT:-dba-sgd-pitr-demo}"
fi

# --auto ejecuta todas las etapas sin pausas (grabacion de respaldo, pruebas).
AUTO=0
[[ "${1:-}" == "--auto" ]] && AUTO=1

# Colores solo en una terminal: al redirigir a un archivo (o con NO_COLOR)
# la salida queda en texto plano.
if [[ -t 1 && -z "${NO_COLOR:-}" ]]; then
    ROJO=$'\033[1;31m'; VERDE=$'\033[1;32m'; AMARILLO=$'\033[1;33m'
    CIAN=$'\033[1;36m'; GRIS=$'\033[0;90m'; NEGRITA=$'\033[1m'; NORMAL=$'\033[0m'
else
    ROJO=""; VERDE=""; AMARILLO=""; CIAN=""; GRIS=""; NEGRITA=""; NORMAL=""
fi

titulo() {
    echo "${CIAN}$1${NORMAL}"
}

ok() {
    echo "${VERDE}[OK]${NORMAL} $1"
}

error() {
    echo "${ROJO}[ERROR] $1${NORMAL}"
}

pausa() {
    if [[ "$AUTO" -eq 0 ]]; then
        read -r -p "${GRIS}$1${NORMAL}"
    fi
}
COMPOSE_FILE="$ROOT_DIR/compose.demo.yaml"
COMPOSE=(docker compose -p "$PROJECT" -f "$COMPOSE_FILE")

SCRIPTS=(
  "02_backup.sh"
  "03_cambios_legitimos.sh"
  "04_incidente.sh"
  "05_registros_posteriores.sh"
  "06_restaurar_pitr.sh"
  "07_reconciliar_post_incidente.sh"
  "08_validar_integridad.sh"
)

NOMBRES=(
  "Respaldo FULL con pgBackRest"
  "Cambios legítimos posteriores al respaldo"
  "Simulación del incidente"
  "Preservación de registros posteriores"
  "Recuperación PITR en instancia independiente"
  "Reconciliación de registros posteriores"
  "Validación de integridad"
)

RESUMENES=(
  "Respaldo FULL ejecutado y verificado."
  "Revisiones legítimas aplicadas después del respaldo FULL (solo vuelven vía WAL)."
  "Foto previa tomada; UPDATE accidental sobre los 100 registros con XID y hora registrados."
  "Tres registros posteriores preservados mediante CSV."
  "PITR hasta justo antes del COMMIT del incidente. Base recuperada en puerto 5433."
  "Tres registros posteriores reincorporados."
  "Los 100 registros recuperados son idénticos a la foto previa al UPDATE."
)

ESTADOS=(
  "PENDIENTE" "PENDIENTE" "PENDIENTE" "PENDIENTE"
  "PENDIENTE" "PENDIENTE" "PENDIENTE"
)

mostrar_checklist() {
    local completadas=0
    local j

    echo
    titulo "========== CHECKLIST DE RECUPERACIÓN =========="

    for j in "${!SCRIPTS[@]}"; do
        case "${ESTADOS[$j]}" in
            COMPLETADO)
                echo "${VERDE}[✓]${NORMAL} ${NOMBRES[$j]}"
                completadas=$((completadas + 1))
                ;;
            "EN CURSO")
                echo "${AMARILLO}[>] ${NOMBRES[$j]}  <-- EN CURSO${NORMAL}"
                ;;
            *)
                echo "${GRIS}[ ] ${NOMBRES[$j]}${NORMAL}"
                ;;
        esac
    done

    echo
    echo "${NEGRITA}Progreso: $completadas/${#SCRIPTS[@]} etapas completadas${NORMAL}"
    titulo "==============================================="
}

fallo() {
    local codigo=$?
    echo
    error "La demostración se interrumpió."
    echo "Código de salida: $codigo"
    echo "El entorno se conserva para diagnóstico."
    exit "$codigo"
}

trap fallo ERR

echo
titulo "=================================================="
echo "${NEGRITA}       DEMOSTRACIÓN DE RECUPERACIÓN PITR${NORMAL}"
echo "       PostgreSQL 17 + pgBackRest"
titulo "=================================================="

# --------------------------------------------------
# PREVALIDACIÓN
# --------------------------------------------------

echo
titulo "========== PREVALIDACIÓN =========="

[[ -f "$COMPOSE_FILE" ]] || {
    error "No existe compose.demo.yaml."
    exit 1
}

[[ -f "$ROOT_DIR/.env" ]] || {
    error "No existe .env."
    exit 1
}

for SCRIPT in "${SCRIPTS[@]}"; do
    [[ -f "$ROOT_DIR/scripts/$SCRIPT" ]] || {
        error "No existe scripts/$SCRIPT."
        exit 1
    }
done

command -v docker >/dev/null 2>&1 || {
    error "Docker no está instalado o no está disponible."
    exit 1
}

docker info >/dev/null

# Comprueba que Compose y las variables sean válidos
"${COMPOSE[@]}" config -q

ok "Archivos, Docker y configuración verificados."

# --------------------------------------------------
# LIMPIEZA INICIAL DEL PROYECTO
# --------------------------------------------------

echo
titulo "========== LIMPIEZA INICIAL =========="
echo "Proyecto exclusivo: $PROJECT"
echo "${AMARILLO}Se eliminarán los contenedores y volúmenes anteriores.${NORMAL}"
echo

pausa "Presiona ENTER para limpiar e iniciar el laboratorio..."

"${COMPOSE[@]}" down --volumes --remove-orphans

echo
ok "Entorno anterior del proyecto eliminado."

# --------------------------------------------------
# PREPARACIÓN DEL LABORATORIO
# --------------------------------------------------

echo
titulo "========== PREPARACIÓN DEL LABORATORIO =========="

"${COMPOSE[@]}" up -d --build postgres

echo
echo "Esperando disponibilidad de PostgreSQL..."

DISPONIBLE=0

for intento in {1..60}; do
    if "${COMPOSE[@]}" exec -T postgres \
        pg_isready -U postgres -d postulaciones_db >/dev/null 2>&1; then
        DISPONIBLE=1
        break
    fi

    sleep 2
done

if [[ "$DISPONIBLE" -ne 1 ]]; then
    error "PostgreSQL no estuvo disponible en 120 segundos."
    exit 1
fi

ok "PostgreSQL principal disponible en puerto 5432."

# --------------------------------------------------
# EJECUCIÓN DE LAS SIETE ETAPAS
# --------------------------------------------------

for i in "${!SCRIPTS[@]}"; do

    SCRIPT="${SCRIPTS[$i]}"
    NOMBRE="${NOMBRES[$i]}"
    RUTA="$ROOT_DIR/scripts/$SCRIPT"

    ESTADOS[$i]="EN CURSO"

    mostrar_checklist

    echo
    titulo "=================================================="
    echo "${AMARILLO}ETAPA $((i + 1))/${#SCRIPTS[@]}: $NOMBRE${NORMAL}"
    echo "${GRIS}Script: $SCRIPT${NORMAL}"
    titulo "=================================================="
    echo

    pausa "Presiona ENTER para ejecutar esta etapa..."

    if bash "$RUTA"; then
        ESTADOS[$i]="COMPLETADO"

        echo
        ok "Etapa $((i + 1)) completada."
        echo "${NEGRITA}RESUMEN:${NORMAL} ${RESUMENES[$i]}"

    else
        echo
        error "Falló la etapa: $NOMBRE."
        echo "Se detiene la demostración."
        echo "No se ejecutará la limpieza."
        exit 1
    fi

    mostrar_checklist

    if (( i < ${#SCRIPTS[@]} - 1 )); then
        pausa "Presiona ENTER para continuar..."
    fi
done

# --------------------------------------------------
# RESULTADO FINAL
# --------------------------------------------------

echo
titulo "=================================================="
echo "${VERDE}          RECUPERACIÓN COMPLETADA${NORMAL}"
titulo "=================================================="

mostrar_checklist

echo
echo "${NEGRITA}RESULTADOS FINALES${NORMAL}"
echo "----------------------------------------------"
echo "PostgreSQL principal : puerto 5432"
echo "PostgreSQL recovery  : puerto 5433"
echo "Registros originales: ${VERDE}100${NORMAL}"
echo "Registros posteriores: ${VERDE}3${NORMAL}"
echo "Total reconciliado   : ${VERDE}103${NORMAL}"
echo "Registros rechazados : ${VERDE}0${NORMAL}"
echo "Integridad           : ${VERDE}VALIDADA${NORMAL}"
echo
echo "Evidencias disponibles en: results/"
echo
echo "Contenedores y volúmenes permanecen activos."
echo "No se ejecutará limpieza al finalizar."
echo
titulo "=================================================="
echo "${NEGRITA}       FIN DE LA DEMOSTRACIÓN PITR${NORMAL}"
titulo "=================================================="

#!/usr/bin/env bash
set -Eeuo pipefail

ROOT_DIR="$(cd "$(dirname "$0")/.." && pwd)"
cd "$ROOT_DIR"

if [[ -f .env ]]; then
  COMPOSE_PROJECT_NAME="$(sed -n 's/^COMPOSE_PROJECT_NAME=//p' .env | tail -n 1)"
fi

COMPOSE_PROJECT_NAME="${COMPOSE_PROJECT_NAME:-dba-sgd-pitr-demo}"
[[ "$COMPOSE_PROJECT_NAME" =~ ^[a-z0-9][a-z0-9_-]*$ ]] || {
  echo "ERROR: COMPOSE_PROJECT_NAME contiene caracteres no permitidos."
  exit 1
}


COMPOSE=(docker compose -p "$COMPOSE_PROJECT_NAME" -f compose.demo.yaml)
mkdir -p results
LOG="results/backup_$(date '+%Y%m%d_%H%M%S').txt"

{
  echo "=== BACKUP COMPLETO PostgreSQL / pgBackRest ==="
  date
  echo

  echo "[1] Verificando PostgreSQL"
  "${COMPOSE[@]}" exec -T postgres pg_isready -U postgres -d postulaciones_db
  echo

  echo "[2] Verificando registros iniciales"
  "${COMPOSE[@]}" exec -T postgres psql -U postgres -d postulaciones_db \
    -c "SELECT COUNT(*) AS total FROM postulaciones;"

  TOTAL="$("${COMPOSE[@]}" exec -T postgres psql -U postgres -d postulaciones_db -Atc "SELECT COUNT(*) FROM postulaciones;")"

  if [[ "$TOTAL" != "100" ]]; then
    echo "ERROR: se esperaban 100 registros y se encontraron $TOTAL"
    exit 1
  fi
  echo

  echo "[3] Verificando stanza pgBackRest"
  if "${COMPOSE[@]}" exec -T -u postgres postgres \
      test -f /var/lib/pgbackrest/backup/main/backup.info; then
    echo "Stanza main existente. Se continúa."
  else
    echo "Stanza main no encontrada. Inicializando automáticamente..."
    "${COMPOSE[@]}" exec -T -u postgres postgres \
      pgbackrest --stanza=main stanza-create
  fi
  echo

  echo "[4] Creando backup FULL"
  "${COMPOSE[@]}" exec -T -u postgres postgres \
    pgbackrest --stanza=main backup --type=full
  echo

  echo "[5] Verificando backup"
  "${COMPOSE[@]}" exec -T -u postgres postgres \
    pgbackrest --stanza=main info
  echo

  echo "BACKUP COMPLETADO CORRECTAMENTE"
  date
} 2>&1 | tee "$LOG"

echo "Evidencia guardada en: $LOG"

  exit 1

COMPOSE=(docker compose -p "$COMPOSE_PROJECT_NAME" -f compose.demo.yaml)
mkdir -p results

# Restore point preventivo: es el plan B de la recuperación. El objetivo
# principal es la transacción del incidente (XID), que se registra en 04.
PUNTO="pitr_objetivo_$(date '+%Y%m%d_%H%M%S')"
LOG="results/cambios_legitimos_$(date '+%Y%m%d_%H%M%S').txt"

{
  echo "=== CAMBIOS LEGITIMOS POSTERIORES AL RESPALDO ==="
  date
  echo
  echo "Estos cambios ocurren despues del respaldo FULL: solo pueden volver"
  echo "si la recuperacion reproduce el WAL archivado, no basta el respaldo base."
  echo

  echo "[1] Estado antes de los cambios"
  "${COMPOSE[@]}" exec -T postgres psql -U postgres -d postulaciones_db \
    -c "SELECT estado, COUNT(*) FROM postulaciones GROUP BY estado ORDER BY estado;"

  echo "[2] Revisores aprueban y preseleccionan postulaciones"
  "${COMPOSE[@]}" exec -T postgres psql -v ON_ERROR_STOP=1 \
    -U postgres -d postulaciones_db <<'SQL'
UPDATE postulaciones
SET estado = 'aprobada',
    observacion = 'Revision legitima: aprobada por revisor'
WHERE id BETWEEN 1 AND 10;

UPDATE postulaciones
SET estado = 'preseleccionada',
    observacion = 'Revision legitima: preseleccionada por revisor'
WHERE id BETWEEN 11 AND 15;
SQL
  echo

  echo "[3] Estado despues de los cambios"
  "${COMPOSE[@]}" exec -T postgres psql -U postgres -d postulaciones_db \
    -c "SELECT estado, COUNT(*) FROM postulaciones GROUP BY estado ORDER BY estado;"

  echo "[4] Creando restore point preventivo (plan B): $PUNTO"
  LSN="$("${COMPOSE[@]}" exec -T postgres psql -U postgres -d postulaciones_db -Atc \
    "SELECT pg_create_restore_point('$PUNTO');")"
  echo "LSN: $LSN"
  printf 'PUNTO=%s\nLSN=%s\n' "$PUNTO" "$LSN" > results/restore_point.env
  echo

  echo "CAMBIOS LEGITIMOS REGISTRADOS"
  date
} 2>&1 | tee "$LOG"

echo "Datos guardados en results/restore_point.env"
echo "Evidencia guardada en: $LOG"
=================== scripts/04_incidente.sh
#!/usr/bin/env bash
set -Eeuo pipefail

ROOT_DIR="$(cd "$(dirname "$0")/.." && pwd)"
cd "$ROOT_DIR"

if [[ -f .env ]]; then
  COMPOSE_PROJECT_NAME="$(sed -n 's/^COMPOSE_PROJECT_NAME=//p' .env | tail -n 1)"
fi

COMPOSE_PROJECT_NAME="${COMPOSE_PROJECT_NAME:-dba-sgd-pitr-demo}"
[[ "$COMPOSE_PROJECT_NAME" =~ ^[a-z0-9][a-z0-9_-]*$ ]] || {
  echo "ERROR: COMPOSE_PROJECT_NAME contiene caracteres no permitidos."
  exit 1
}


COMPOSE=(docker compose -p "$COMPOSE_PROJECT_NAME" -f compose.demo.yaml)
SQL="scripts/simular_incidente.sql"

mkdir -p results

[[ -f results/restore_point.env ]] || {
  echo "ERROR: no existe results/restore_point.env (ejecute el script 03)."
  exit 1
}

[[ -f "$SQL" ]] || {
  echo "ERROR: no existe $SQL"
  exit 1
}

echo "[1] Validando estado previo del laboratorio"

ESTADO="$("${COMPOSE[@]}" exec -T postgres psql \
  -v ON_ERROR_STOP=1 -U postgres -d postulaciones_db -Atc "
    SELECT
      COUNT(*),
      COUNT(*) FILTER (
        WHERE estado = 'rechazada'
          AND observacion = 'INCIDENTE: UPDATE accidental sin WHERE'
      ),
      COUNT(*) FILTER (
        WHERE observacion = 'POST_INCIDENTE_RECUPERABLE'
      )
    FROM postulaciones;
  ")"

IFS='|' read -r TOTAL AFECTADOS POSTERIORES <<< "$ESTADO"

if [[ "$TOTAL" != "100" || "$AFECTADOS" != "0" || "$POSTERIORES" != "0" ]]; then
  echo "ERROR: estado no válido para ejecutar el incidente."
  echo "Total: $TOTAL | Afectados: $AFECTADOS | Posteriores: $POSTERIORES"
  echo "Se cancela para proteger los datos."
  exit 1
fi

RUN_ID="$(date '+%Y%m%d_%H%M%S')"
LOG="results/incidente_${RUN_ID}.txt"
SNAPSHOT="results/antes_incidente_${RUN_ID}.csv"

{
  echo "=== SIMULACION DEL INCIDENTE ==="
  date
  echo

  # Cambio legitimo segundos antes del incidente: si la recuperacion queda
  # "justo antes del UPDATE", este cambio debe volver; si queda antes, se pierde.
  echo "[2] Ultimo cambio legitimo antes del incidente (id 50)"
  "${COMPOSE[@]}" exec -T postgres psql -v ON_ERROR_STOP=1 \
    -U postgres -d postulaciones_db -c "
      UPDATE postulaciones
      SET estado = 'aprobada',
          observacion = 'Revision legitima: aprobada segundos antes del incidente'
      WHERE id = 50;"
  echo

  echo "[3] Foto del estado previo (todas las columnas): $SNAPSHOT"
  "${COMPOSE[@]}" exec -T postgres psql -v ON_ERROR_STOP=1 \
    -U postgres -d postulaciones_db \
    -c "\copy (SELECT id, nombre_postulante, cargo, region, estado, creado_en, observacion FROM postulaciones ORDER BY id) TO STDOUT WITH CSV HEADER" \
    > "$SNAPSHOT"
  "${COMPOSE[@]}" exec -T postgres psql -U postgres -d postulaciones_db \
    -c "SELECT estado, COUNT(*) FROM postulaciones GROUP BY estado ORDER BY estado;"

  sleep 2

  echo "[4] Ejecutando UPDATE accidental (hora del servidor y XID registrados)"
  SALIDA="$("${COMPOSE[@]}" exec -T postgres psql -X -At -v ON_ERROR_STOP=1 \
    -U postgres -d postulaciones_db < "$SQL")"
  echo "$SALIDA"

  XID_INCIDENTE="$(sed -n 's/^XID_INCIDENTE=//p' <<< "$SALIDA")"
  HORA_INCIDENTE="$(sed -n 's/^HORA_INCIDENTE=//p' <<< "$SALIDA")"

  if [[ -z "$XID_INCIDENTE" || -z "$HORA_INCIDENTE" ]]; then
    echo "ERROR: no se pudo registrar el XID o la hora del incidente."
    exit 1
  fi

  printf 'XID_INCIDENTE=%s\nHORA_INCIDENTE="%s"\nSNAPSHOT=%s\n' \
    "$XID_INCIDENTE" "$HORA_INCIDENTE" "$SNAPSHOT" >> results/restore_point.env
  echo

  echo "[5] Verificando impacto"
  "${COMPOSE[@]}" exec -T postgres psql -U postgres -d postulaciones_db \
    -c "SELECT estado, COUNT(*) FROM postulaciones GROUP BY estado ORDER BY estado;"
  echo
  echo "INCIDENTE SIMULADO"
  echo "XID del incidente : $XID_INCIDENTE"
  echo "Hora del incidente: $HORA_INCIDENTE"
  date
} 2>&1 | tee "$LOG"

echo "Evidencia guardada en: $LOG"
=================== scripts/05_registros_posteriores.sh
#!/usr/bin/env bash
set -Eeuo pipefail

ROOT_DIR="$(cd "$(dirname "$0")/.." && pwd)"
cd "$ROOT_DIR"

if [[ -f .env ]]; then
  COMPOSE_PROJECT_NAME="$(sed -n 's/^COMPOSE_PROJECT_NAME=//p' .env | tail -n 1)"
fi

COMPOSE_PROJECT_NAME="${COMPOSE_PROJECT_NAME:-dba-sgd-pitr-demo}"
[[ "$COMPOSE_PROJECT_NAME" =~ ^[a-z0-9][a-z0-9_-]*$ ]] || {
  echo "ERROR: COMPOSE_PROJECT_NAME contiene caracteres no permitidos."
  exit 1
}


COMPOSE=(docker compose -p "$COMPOSE_PROJECT_NAME" -f compose.demo.yaml)
mkdir -p results

RUN_ID="$(date '+%Y%m%d_%H%M%S')"
LOG="results/post_incidente_${RUN_ID}.txt"
CSV="results/post_incidente_${RUN_ID}.csv"
CSV_PATH_FILE="results/post_incidente_csv.path"

{
  echo "=== REGISTROS LEGITIMOS POSTERIORES AL INCIDENTE ==="
  date
  echo

  echo "[1] Validando estado del incidente y registros previos"

  ESTADO="$("${COMPOSE[@]}" exec -T postgres psql \
    -v ON_ERROR_STOP=1 -U postgres -d postulaciones_db -Atc "
      SELECT
        COUNT(*),
        COUNT(*) FILTER (
          WHERE estado = 'rechazada'
            AND observacion = 'INCIDENTE: UPDATE accidental sin WHERE'
        ),
        COUNT(*) FILTER (
          WHERE observacion = 'POST_INCIDENTE_RECUPERABLE'
        )
      FROM postulaciones;
    ")"

  IFS='|' read -r TOTAL AFECTADOS POSTERIORES <<< "$ESTADO"

  if [[ "$TOTAL" != "100" || "$AFECTADOS" != "100" || "$POSTERIORES" != "0" ]]; then
    echo "ERROR: estado no válido para insertar registros posteriores."
    echo "Total: $TOTAL | Afectados: $AFECTADOS | Posteriores: $POSTERIORES"
    exit 1
  fi

  if [[ -e "$CSV" ]]; then
    echo "ERROR: ya existe $CSV. Se cancela para proteger la evidencia."
    exit 1
  fi

  echo
  echo "[2] Insertando tres postulaciones legítimas"
  "${COMPOSE[@]}" exec -T postgres psql -v ON_ERROR_STOP=1 \
    -U postgres -d postulaciones_db <<'SQL'
INSERT INTO postulaciones
    (nombre_postulante, cargo, region, estado, observacion)
VALUES
    ('Registro Posterior 1', 'Administrador de Bases de Datos',
     'Metropolitana', 'en_revision', 'POST_INCIDENTE_RECUPERABLE'),
    ('Registro Posterior 2', 'Analista de Datos',
     'Valparaiso', 'preseleccionada', 'POST_INCIDENTE_RECUPERABLE'),
    ('Registro Posterior 3', 'Ingeniero de Infraestructura',
     'Biobio', 'pendiente', 'POST_INCIDENTE_RECUPERABLE');
SQL

  echo
  echo "[3] Exportando registros posteriores a CSV"
  "${COMPOSE[@]}" exec -T postgres psql -U postgres -d postulaciones_db \
    -c "\copy (SELECT nombre_postulante, cargo, region, estado, creado_en, observacion FROM postulaciones WHERE observacion = 'POST_INCIDENTE_RECUPERABLE' ORDER BY nombre_postulante) TO STDOUT WITH CSV HEADER" \
    > "$CSV"

  FILAS_CSV=$(($(wc -l < "$CSV") - 1))
  if [[ "$FILAS_CSV" -ne 3 ]]; then
    echo "ERROR: el CSV debe contener exactamente 3 registros."
    exit 1
  fi

  printf '%s\n' "$CSV" > "$CSV_PATH_FILE"

  echo
  echo "[4] Verificando registros"
  "${COMPOSE[@]}" exec -T postgres psql -U postgres -d postulaciones_db \
    -c "SELECT COUNT(*) AS total, COUNT(*) FILTER (WHERE estado = 'rechazada') AS rechazadas, COUNT(*) FILTER (WHERE observacion = 'POST_INCIDENTE_RECUPERABLE') AS posteriores FROM postulaciones;"

  echo
  echo "[5] Forzando cambio de segmento WAL"
  "${COMPOSE[@]}" exec -T postgres psql \
    -v ON_ERROR_STOP=1 -U postgres -d postulaciones_db \
    -c "SELECT pg_switch_wal();"

  echo "Solicitud de cambio de segmento WAL completada."
  echo

  # pg_switch_wal() no espera al archivador. pgbackrest check fuerza otro
  # cambio de segmento y espera a que llegue al repositorio (o falla); como
  # el archivado es en orden, garantiza que el WAL del incidente ya esta ahi.
  echo "[6] Verificando que el WAL archivado esté en el repositorio"
  "${COMPOSE[@]}" exec -T -u postgres postgres \
    pgbackrest --stanza=main check
  "${COMPOSE[@]}" exec -T postgres psql -U postgres -d postulaciones_db \
    -c "SELECT archived_count, last_archived_wal, last_archived_time, failed_count, last_failed_wal FROM pg_stat_archiver;"

  echo "[7] Contenido del CSV"
  cat "$CSV"

  echo
  echo "REGISTROS POSTERIORES Y EXPORTACION COMPLETADOS"
  date
} 2>&1 | tee "$LOG"

echo "CSV generado: $CSV"
echo "Referencia guardada en: $CSV_PATH_FILE"
echo "Evidencia guardada en: $LOG"
=================== scripts/06_restaurar_pitr.sh
#!/usr/bin/env bash
set -Eeuo pipefail

ROOT_DIR="$(cd "$(dirname "$0")/.." && pwd)"
cd "$ROOT_DIR"

if [[ -f .env ]]; then
  COMPOSE_PROJECT_NAME="$(sed -n 's/^COMPOSE_PROJECT_NAME=//p' .env | tail -n 1)"
fi

COMPOSE_PROJECT_NAME="${COMPOSE_PROJECT_NAME:-dba-sgd-pitr-demo}"
[[ "$COMPOSE_PROJECT_NAME" =~ ^[a-z0-9][a-z0-9_-]*$ ]] || {
  echo "ERROR: COMPOSE_PROJECT_NAME contiene caracteres no permitidos."
  exit 1
}


COMPOSE=(docker compose -p "$COMPOSE_PROJECT_NAME" -f compose.demo.yaml)

RESTORE_ENV="results/restore_point.env"

if [[ ! -s "$RESTORE_ENV" ]]; then
  echo "ERROR: no existe el archivo del punto de recuperación o está vacío."
  exit 1
fi

source "$RESTORE_ENV"

# Objetivo de recuperacion (TARGET_TYPE):
#   xid  (por defecto) justo antes del COMMIT de la transaccion del incidente.
#   time hora del servidor al iniciar la transaccion del incidente.
#   name restore point preventivo del script 03 (plan B: pierde lo posterior).
TARGET_TYPE="${TARGET_TYPE:-xid}"

case "$TARGET_TYPE" in
  xid)
    TARGET="${XID_INCIDENTE:-}"
    TARGET_ARGS=(--type=xid --target="$TARGET" --target-exclusive)
    ;;
  time)
    TARGET="${HORA_INCIDENTE:-}"
    TARGET_ARGS=(--type=time --target="$TARGET" --target-exclusive)
    ;;
  name)
    TARGET="${PUNTO:-}"
    TARGET_ARGS=(--type=name --target="$TARGET")
    ;;
  *)
    echo "ERROR: TARGET_TYPE debe ser xid, time o name."
    exit 1
    ;;
esac

if [[ -z "$TARGET" ]]; then
  echo "ERROR: el objetivo '$TARGET_TYPE' no está definido en $RESTORE_ENV."
  exit 1
fi

LOG="results/restauracion_pitr_$(date '+%Y%m%d_%H%M%S').txt"
INICIO=0

{
  echo "=== RESTAURACION PITR POSTGRESQL 17 ==="
  date
  echo "Tipo de objetivo: $TARGET_TYPE"
  echo "Objetivo: $TARGET"
  echo

  echo "[1] Comprobando disponibilidad de la base principal"
  "${COMPOSE[@]}" exec -T postgres \
    pg_isready -U postgres -d postulaciones_db

  echo "Base principal disponible."
  echo
  echo "[2] Comprobando que recovery esté detenido"

  if "${COMPOSE[@]}" ps --status running --services | grep -qx recovery; then
    echo "ERROR: recovery está ejecutándose. Se cancela para proteger los datos."
    exit 1
  fi

  echo "Recovery detenido."
  echo
  echo "[3] Comprobando destino de restauracion"
  "${COMPOSE[@]}" run --rm --no-deps --entrypoint sh recovery -c \
    'if [ -n "$(ls -A /var/lib/postgresql/data)" ]; then echo "ERROR: el volumen de recuperacion no esta vacio."; exit 1; fi; chown postgres:postgres /var/lib/postgresql/data'
  echo "Destino vacio y permisos preparados."
  echo

  echo "[4] Ejecutando restauracion PITR"
  INICIO="$(date +%s)"
  "${COMPOSE[@]}" run --rm --no-deps --user postgres \
    --entrypoint pgbackrest recovery \
    --stanza=main \
    "${TARGET_ARGS[@]}" \
    --target-action=promote \
    restore
  echo

  echo "[5] Iniciando PostgreSQL restaurado en puerto 5433"
  "${COMPOSE[@]}" up -d recovery

  # pg_isready acepta conexiones durante la recuperacion (hot standby):
  # la base solo esta disponible para escribir cuando termina y se promueve.
  echo "[6] Esperando fin de la recuperacion y promocion"
  LISTO=0
  for intento in $(seq 1 60); do
    if [[ "$("${COMPOSE[@]}" exec -T recovery psql -U postgres -d postulaciones_db \
          -Atc "SELECT pg_is_in_recovery();" 2>/dev/null)" == "f" ]]; then
      LISTO=1
      break
    fi
    sleep 1
  done

  if [[ "$LISTO" != "1" ]]; then
    echo "ERROR: PostgreSQL no terminó la recuperación dentro del tiempo esperado."
    "${COMPOSE[@]}" logs --tail 30 recovery
    exit 1
  fi

  FIN="$(date +%s)"
  RTO=$((FIN - INICIO))
  echo "Recuperacion terminada y base promovida."

  echo
  echo "[7] Verificando datos restaurados"
  "${COMPOSE[@]}" exec -T recovery psql -U postgres -d postulaciones_db \
    -c "SELECT estado, COUNT(*) FROM postulaciones GROUP BY estado ORDER BY estado;" \
    -c "SELECT id, estado, observacion FROM postulaciones WHERE id = 50;"

  echo
  echo "RESTAURACION PITR COMPLETADA"
  echo "Inicio epoch: $INICIO"
  echo "Fin epoch: $FIN"
  echo "RTO desde inicio de restauracion hasta base promovida: ${RTO} segundos"
  date
} 2>&1 | tee "$LOG"

echo "Evidencia guardada en: $LOG"

[[ -s "$CSV_PATH_FILE" ]] || {
  echo "ERROR: no existe la referencia al CSV generado por el script 05."
  exit 1
}

CSV="$(cat "$CSV_PATH_FILE")"

[[ -s "$CSV" ]] || {
  echo "ERROR: CSV inexistente o vacío: $CSV"
  exit 1
}

FILAS_CSV=$(($(wc -l < "$CSV") - 1))

  echo "ERROR: el CSV debe contener exactamente 3 registros."
  echo "Registros encontrados: $FILAS_CSV"
  exit 1
fi

  echo "=== RECONCILIACION DE REGISTROS POSTERIORES ==="
  date
  echo
  echo "CSV utilizado: $CSV"

  EXISTENTES="$("${COMPOSE[@]}" exec -T recovery psql -U postgres -d postulaciones_db -Atc \
    "SELECT COUNT(*) FROM postulaciones WHERE observacion = 'POST_INCIDENTE_RECUPERABLE';")"

  if [[ "$EXISTENTES" != "0" ]]; then
    echo "ERROR: ya existen registros posteriores en recovery. Se cancela."
    exit 1
  fi

  TOTAL_RECOVERY="$("${COMPOSE[@]}" exec -T recovery psql \
    -v ON_ERROR_STOP=1 -U postgres -d postulaciones_db -Atc \
    "SELECT COUNT(*) FROM postulaciones;")"

  if [[ "$TOTAL_RECOVERY" != "100" ]]; then
    echo "ERROR: recovery debe contener exactamente 100 registros antes de reconciliar."
    echo "Registros encontrados: $TOTAL_RECOVERY"
    exit 1
  fi

  echo "Recovery validado: 100 registros originales."

  echo
  echo "[1] Importando CSV en recovery (puerto 5433)"
  "${COMPOSE[@]}" exec -T recovery psql -v ON_ERROR_STOP=1 \
    -U postgres -d postulaciones_db \
    -c "\copy postulaciones (nombre_postulante, cargo, region, estado, creado_en, observacion) FROM STDIN WITH CSV HEADER" \
    < "$CSV"

  echo
  echo "[2] Validando resultado"
  "${COMPOSE[@]}" exec -T recovery psql -U postgres -d postulaciones_db \
    -c "SELECT COUNT(*) AS total, COUNT(*) FILTER (WHERE estado = 'rechazada') AS rechazadas, COUNT(*) FILTER (WHERE observacion = 'POST_INCIDENTE_RECUPERABLE') AS posteriores FROM postulaciones;"

  echo
  echo "[3] Verificando los tres registros reincorporados"
  "${COMPOSE[@]}" exec -T recovery psql -U postgres -d postulaciones_db \
    -c "SELECT nombre_postulante, cargo, region, estado, creado_en FROM postulaciones WHERE observacion = 'POST_INCIDENTE_RECUPERABLE' ORDER BY nombre_postulante;"

  TOTAL="$("${COMPOSE[@]}" exec -T recovery psql -U postgres -d postulaciones_db -Atc "SELECT COUNT(*) FROM postulaciones;")"
  POSTERIORES="$("${COMPOSE[@]}" exec -T recovery psql -U postgres -d postulaciones_db -Atc "SELECT COUNT(*) FROM postulaciones WHERE observacion = 'POST_INCIDENTE_RECUPERABLE';")"
  RECHAZADAS="$("${COMPOSE[@]}" exec -T recovery psql -U postgres -d postulaciones_db -Atc "SELECT COUNT(*) FROM postulaciones WHERE estado = 'rechazada';")"

  [[ "$TOTAL" == "103" && "$POSTERIORES" == "3" && "$RECHAZADAS" == "0" ]] || {
    echo "ERROR: la validación final no coincide con lo esperado."
    exit 1
  }

  echo
  echo "RECONCILIACION COMPLETADA: 103 registros, 3 posteriores y 0 rechazados."
  date
} 2>&1 | tee "$LOG"

echo "Evidencia guardada en: $LOG"

=================== scripts/simular_incidente.sql
-- Proyecto: Laboratorio DBA - PITR PostgreSQL
-- Proposito: Simular un UPDATE accidental sin WHERE.
-- ADVERTENCIA: modifica todos los registros de postulaciones.
-- Ejecutar solo en el entorno de prueba preparado.
--
-- Se registran el XID y la hora de la transaccion del incidente: son el
-- objetivo de la recuperacion PITR (se restaura justo antes de su COMMIT).

BEGIN;

SELECT 'XID_INCIDENTE=' || pg_current_xact_id();
SELECT 'HORA_INCIDENTE=' || clock_timestamp();

UPDATE postulaciones
SET estado = 'rechazada',
    observacion = 'INCIDENTE: UPDATE accidental sin WHERE';

SELECT 'AFECTADOS=' || COUNT(*) FILTER (
    WHERE estado = 'rechazada'
      AND observacion = 'INCIDENTE: UPDATE accidental sin WHERE'
)
FROM postulaciones;

COMMIT;
