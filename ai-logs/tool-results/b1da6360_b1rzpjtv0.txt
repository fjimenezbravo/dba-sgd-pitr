=================== sql/01_schema.sql
     1	-- Proyecto: Recuperacion PITR con PostgreSQL 17 y pgBackRest
     2	-- Proposito: Definir la tabla de postulaciones para la prueba tecnica.
     3	
     4	CREATE TABLE IF NOT EXISTS postulaciones (
     5	    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
     6	    nombre_postulante VARCHAR(100) NOT NULL,
     7	    cargo VARCHAR(150) NOT NULL,
     8	    region VARCHAR(50) NOT NULL,
     9	    estado VARCHAR(30) NOT NULL DEFAULT 'en_revision',
    10	    creado_en TIMESTAMPTZ NOT NULL DEFAULT clock_timestamp(),
    11	    observacion TEXT
    12	);
    13	
    14	COMMENT ON TABLE postulaciones IS
    15	    'Tabla ficticia para demostrar respaldo, incidente y recuperacion PITR';

=================== scripts/06_restaurar_pitr.sh
     1	#!/usr/bin/env bash
     2	set -Eeuo pipefail
     3	
     4	ROOT_DIR="$(cd "$(dirname "$0")/.." && pwd)"
     5	cd "$ROOT_DIR"
     6	
     7	if [[ -f .env ]]; then
     8	  COMPOSE_PROJECT_NAME="$(sed -n 's/^COMPOSE_PROJECT_NAME=//p' .env | tail -n 1)"
     9	fi
    10	
    11	COMPOSE_PROJECT_NAME="$COMPOSE_PROJECT_NAME"
    12	[[ "$COMPOSE_PROJECT_NAME" =~ ^[a-z0-9][a-z0-9_-]*$ ]] || {
    13	  echo "ERROR: COMPOSE_PROJECT_NAME contiene caracteres no permitidos."
    14	  exit 1
    15	}
    16	
    17	
    18	COMPOSE=(docker compose -p "$COMPOSE_PROJECT_NAME" -f compose.demo.yaml)
    19	
    20	RESTORE_ENV="results/restore_point.env"
    21	
    22	if [[ ! -s "$RESTORE_ENV" ]]; then
    23	  echo "ERROR: no existe el archivo del punto de recuperación o está vacío."
    24	  exit 1
    25	fi
    26	
    27	source "$RESTORE_ENV"
    28	
    29	if [[ -z "${PUNTO:-}" || -z "${LSN:-}" ]]; then
    30	  echo "ERROR: PUNTO o LSN no están definidos en $RESTORE_ENV."
    31	  exit 1
    32	fi
    33	LOG="results/restauracion_pitr_$(date '+%Y%m%d_%H%M%S').txt"
    34	INICIO=0
    35	
    36	{
    37	  echo "=== RESTAURACION PITR POSTGRESQL 17 ==="
    38	  date
    39	  echo "Restore point: $PUNTO"
    40	  echo "LSN: $LSN"
    41	  echo
    42	
    43	  echo "[1] Comprobando disponibilidad de la base principal"
    44	  "${COMPOSE[@]}" exec -T postgres \
    45	    pg_isready -U postgres -d postulaciones_db
    46	
    47	  echo "Base principal disponible."
    48	  echo
    49	  echo "[2] Comprobando que recovery esté detenido"
    50	
    51	  if "${COMPOSE[@]}" ps --status running --services | grep -qx recovery; then
    52	    echo "ERROR: recovery está ejecutándose. Se cancela para proteger los datos."
    53	    exit 1
    54	  fi
    55	
    56	  echo "Recovery detenido."
    57	  echo
    58	  echo "[3] Comprobando destino de restauracion"
    59	  "${COMPOSE[@]}" run --rm --no-deps --entrypoint sh recovery -c \
    60	    'if [ -n "$(ls -A /var/lib/postgresql/data)" ]; then echo "ERROR: el volumen de recuperacion no esta vacio."; exit 1; fi; chown postgres:postgres /var/lib/postgresql/data'
    61	  echo "Destino vacio y permisos preparados."
    62	  echo
    63	
    64	  echo "[4] Ejecutando restauracion PITR"
    65	  INICIO="$(date +%s)"
    66	  "${COMPOSE[@]}" run --rm --no-deps --user postgres \
    67	    --entrypoint pgbackrest recovery \
    68	    --stanza=main \
    69	    --type=name \
    70	    --target="$PUNTO" \
    71	    --target-action=promote \
    72	    restore
    73	  echo
    74	
    75	  echo "[5] Iniciando PostgreSQL restaurado en puerto 5433"
    76	  "${COMPOSE[@]}" up -d recovery
    77	
    78	  echo "[6] Esperando disponibilidad de PostgreSQL"
    79	  LISTO=0
    80	  for intento in $(seq 1 60); do
    81	    if "${COMPOSE[@]}" exec -T recovery pg_isready -U postgres -d postulaciones_db; then
    82	      LISTO=1
    83	      break
    84	    fi
    85	    sleep 2
    86	  done
    87	
    88	  if [[ "$LISTO" != "1" ]]; then
    89	    echo "ERROR: PostgreSQL no quedo disponible dentro del tiempo esperado."
    90	    exit 1
    91	  fi
    92	
    93	  FIN="$(date +%s)"
    94	  RTO=$((FIN - INICIO))
    95	
    96	  echo
    97	  echo "[7] Verificando datos restaurados"
    98	  "${COMPOSE[@]}" exec -T recovery psql -U postgres -d postulaciones_db \
    99	    -c "SELECT COUNT(*) AS total, COUNT(*) FILTER (WHERE estado = 'rechazada') AS rechazadas FROM postulaciones;"
   100	
   101	  echo
   102	  echo "RESTAURACION PITR COMPLETADA"
   103	  echo "Inicio epoch: $INICIO"
   104	  echo "Fin epoch: $FIN"
   105	  echo "RTO hasta disponibilidad: ${RTO} segundos"
   106	  date
   107	} 2>&1 | tee "$LOG"
   108	
   109	echo "Evidencia guardada en: $LOG"

    12	    FROM origen.postulaciones

    24	)

    27	    (SELECT COUNT(*) FROM recuperacion),
